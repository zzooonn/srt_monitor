import uuid
from datetime import date, timedelta

from pydantic import BaseModel, Field, field_validator, model_validator

from .stations import SRT_STATION_NAMES


def _tomorrow() -> str:
    return (date.today() + timedelta(days=1)).strftime("%Y%m%d")


class PublicConfig(BaseModel):
    """민감 필드를 제외한 공개 설정. API 경계에서 사용."""
    departure_date: str = Field(default_factory=_tomorrow, pattern=r"^\d{8}$")
    departure_time: str = Field(default="200000", pattern=r"^\d{6}$")
    max_departure_time: str = Field(default="230000", pattern=r"^\d{6}$")
    departure_station: str = "오송"
    arrival_station: str = "동탄"

    check_special: bool = True
    check_normal: bool = True

    adult_count: int = Field(default=2, ge=0, le=4)
    child_count: int = Field(default=0, ge=0, le=4)
    senior_count: int = Field(default=0, ge=0, le=4)
    disability_1_to_3_count: int = Field(default=0, ge=0, le=4)
    disability_4_to_6_count: int = Field(default=0, ge=0, le=4)

    require_same_row: bool = False
    require_same_car: bool = False
    require_adjacent_seats: bool = True
    prefer_window_seat: bool = True
    require_window_seat: bool = False
    seat_column_order: str = "ABCD"
    adjacent_seat_pairs: list[str] = Field(default_factory=lambda: ["AB", "CD"])
    window_seat_letters: list[str] = Field(default_factory=lambda: ["A", "D"])

    auto_reserve: bool = True
    max_reservations: int = Field(default=1, ge=1, le=20)
    auto_cancel_if_seat_mismatch: bool = True
    notify_if_seat_mismatch: bool = True

    poll_interval: int = Field(default=30, ge=10, le=600)
    heartbeat_interval: int = Field(default=1800, ge=60, le=86400)

    @field_validator("departure_station", "arrival_station")
    @classmethod
    def validate_station(cls, value: str) -> str:
        if value not in SRT_STATION_NAMES:
            raise ValueError(f"지원하지 않는 SRT 역입니다: {value}")
        return value

    @field_validator("departure_time", "max_departure_time")
    @classmethod
    def validate_quarter_hour(cls, value: str) -> str:
        minutes = int(value[2:4])
        seconds = int(value[4:6])
        if seconds != 0 or minutes not in {0, 15, 30, 45}:
            raise ValueError("조회 시간은 00분, 15분, 30분, 45분 단위만 가능합니다.")
        return value

    @model_validator(mode="after")
    def validate_time_range(self) -> "PublicConfig":
        if self.max_departure_time < self.departure_time:
            raise ValueError(
                f"마지막 출발 시간이 시작 출발 시간보다 빠릅니다. "
                f"(시작: {self.departure_time[:2]}:{self.departure_time[2:4]}, "
                f"마지막: {self.max_departure_time[:2]}:{self.max_departure_time[2:4]})"
            )
        return self


class MonitorConfig(PublicConfig):
    """민감 필드 포함 전체 설정. MonitorService 내부 전용."""
    telegram_bot_token: str = ""
    telegram_chat_id: str = ""
    srt_id: str = ""
    srt_password: str = ""

    @property
    def passenger_count(self) -> int:
        total = (
            self.adult_count
            + self.child_count
            + self.senior_count
            + self.disability_1_to_3_count
            + self.disability_4_to_6_count
        )
        return max(1, total)

    @model_validator(mode="after")
    def validate_total_passengers(self) -> "MonitorConfig":
        if self.passenger_count > 4:
            raise ValueError("총 인원은 최대 4명까지 가능합니다.")
        return self


class SecretsPayload(BaseModel):
    """민감 필드 저장용. POST /api/config/secrets 요청 바디."""
    srt_id: str = ""
    srt_password: str = ""
    telegram_bot_token: str = ""
    telegram_chat_id: str = ""


class SecretsStatus(BaseModel):
    """민감 필드 설정 여부. 값 자체는 포함하지 않는다."""
    srt_id_set: bool
    srt_password_set: bool
    telegram_bot_token_set: bool
    telegram_chat_id_set: bool


class MonitorStatus(BaseModel):
    running: bool
    reserved_count: int
    last_message: str
    started_at: str | None = None


class EventItem(BaseModel):
    id: str = Field(default_factory=lambda: uuid.uuid4().hex[:8])
    level: str
    message: str
    timestamp: str
```

import json
from datetime import date, datetime, timedelta
from pathlib import Path

from .schemas import MonitorConfig, PublicConfig, SecretsPayload, SecretsStatus, _tomorrow


ROOT_DIR = Path(__file__).resolve().parents[1]
CONFIG_PATH = ROOT_DIR / "web_config.json"

_SECRETS_FIELDS = {"srt_id", "srt_password", "telegram_bot_token", "telegram_chat_id"}


def _is_past_yyyymmdd(value: str) -> bool:
    try:
        return datetime.strptime(value, "%Y%m%d").date() < date.today()
    except ValueError:
        return True


def _read_raw() -> dict:
    if not CONFIG_PATH.exists():
        return {}
    with CONFIG_PATH.open("r", encoding="utf-8") as f:
        return json.load(f)


def _write_raw(data: dict) -> None:
    with CONFIG_PATH.open("w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)


def load_config() -> MonitorConfig:
    data = _read_raw()
    if _is_past_yyyymmdd(str(data.get("departure_date", ""))):
        data["departure_date"] = _tomorrow()
    return MonitorConfig(**data)


def save_config(config: MonitorConfig) -> MonitorConfig:
    _write_raw(config.model_dump())
    return config


def load_public_config() -> PublicConfig:
    data = _read_raw()
    if _is_past_yyyymmdd(str(data.get("departure_date", ""))):
        data["departure_date"] = _tomorrow()
    public_data = {k: v for k, v in data.items() if k not in _SECRETS_FIELDS}
    return PublicConfig(**public_data)


def load_secrets_status() -> SecretsStatus:
    data = _read_raw()
    return SecretsStatus(
        srt_id_set=bool(str(data.get("srt_id", "")).strip()),
        srt_password_set=bool(str(data.get("srt_password", "")).strip()),
        telegram_bot_token_set=bool(str(data.get("telegram_bot_token", "")).strip()),
        telegram_chat_id_set=bool(str(data.get("telegram_chat_id", "")).strip()),
    )


def save_public_config(public: PublicConfig) -> PublicConfig:
    data = _read_raw()
    secrets = {k: data.get(k, "") for k in _SECRETS_FIELDS}
    merged = {**public.model_dump(), **secrets}
    _write_raw(merged)
    return public


def save_secrets(secrets: SecretsPayload) -> SecretsStatus:
    data = _read_raw()
    for field in _SECRETS_FIELDS:
        value = str(getattr(secrets, field, "") or "")
        if value.strip():
            data[field] = value
    _write_raw(data)
    return load_secrets_status()
```

import asyncio
import json
from contextlib import asynccontextmanager
from typing import Annotated

from fastapi import Depends, FastAPI, Request
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse, StreamingResponse

from .config_store import (
    load_config,
    load_public_config,
    load_secrets_status,
    save_public_config,
    save_secrets,
)
from .monitor_service import MonitorService
from .schemas import PublicConfig, SecretsPayload, SecretsStatus
from .stations import SRT_STATIONS


@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.service = MonitorService()
    yield
    await app.state.service.stop()


app = FastAPI(title="SRT Monitor", lifespan=lifespan)

LOOPBACK_CLIENTS = {"127.0.0.1", "::1", "localhost"}


@app.middleware("http")
async def require_loopback_client(request: Request, call_next):
    client_host = request.client.host if request.client else ""
    if client_host not in LOOPBACK_CLIENTS:
        return JSONResponse(
            status_code=403,
            content={"detail": "SRT Monitor API는 이 PC의 로컬 브라우저에서만 사용할 수 있습니다."},
        )
    return await call_next(request)


app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://127.0.0.1:5173", "http://localhost:5173"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


def get_service(request: Request) -> MonitorService:
    return request.app.state.service


Service = Annotated[MonitorService, Depends(get_service)]


@app.get("/api/config/public")
def get_public_config() -> PublicConfig:
    return load_public_config()


@app.post("/api/config/public")
def update_public_config(public: PublicConfig) -> PublicConfig:
    return save_public_config(public)


@app.get("/api/config/secrets")
def get_secrets_status() -> SecretsStatus:
    return load_secrets_status()


@app.post("/api/config/secrets")
def update_secrets(secrets: SecretsPayload) -> SecretsStatus:
    return save_secrets(secrets)


@app.get("/api/status")
def get_status(service: Service):
    return service.status()


@app.get("/api/stations")
def get_stations():
    return SRT_STATIONS


@app.get("/api/events")
def get_events(service: Service):
    return service.recent_events()


@app.post("/api/start")
async def start_monitor(public: PublicConfig, service: Service):
    save_public_config(public)
    full_config = load_config()
    return await service.start(full_config)


@app.post("/api/stop")
async def stop_monitor(service: Service):
    return await service.stop()


@app.get("/api/events/stream")
async def stream_events(request: Request, service: Service, skip: int = 0):
    async def event_generator():
        resume_id = request.headers.get("last-event-id")
        last_id: str | None = None

        events = service.recent_events()
        if resume_id:
            id_map = {e.id: i for i, e in enumerate(events)}
            pos = id_map.get(resume_id)
            initial = events[pos + 1:] if pos is not None else events
        else:
            initial = events[min(skip, len(events)):]

        for item in initial:
            yield f"id: {item.id}\ndata: {json.dumps(item.model_dump(), ensure_ascii=False)}\n\n"
            last_id = item.id

        while True:
            if await request.is_disconnected():
                break
            events = service.recent_events()
            if last_id is None:
                new_events = events
            else:
                index_map = {item.id: i for i, item in enumerate(events)}
                pos = index_map.get(last_id)
                new_events = events[pos + 1:] if pos is not None else events
            for item in new_events:
                yield f"id: {item.id}\ndata: {json.dumps(item.model_dump(), ensure_ascii=False)}\n\n"
                last_id = item.id
            await asyncio.sleep(1)

    return StreamingResponse(event_generator(), media_type="text/event-stream")

async function request<T>(url: string, options?: RequestInit): Promise<T> {
  const response = await fetch(url, {
    headers: { 'Content-Type': 'application/json' },
    ...options,
  });

  if (!response.ok) {
    const text = await response.text();
    if (response.status === 500) {
      throw new Error(
        '백엔드 서버 오류입니다. Python 설치와 run_backend.bat 창의 에러 메시지를 확인하세요.',
      );
    }
    throw new Error(text || `Request failed: ${response.status}`);
  }

  return response.json() as Promise<T>;
}

export function getPublicConfig() {
  return request<PublicConfig>('/api/config/public');
}

export function savePublicConfig(config: PublicConfig) {
  return request<PublicConfig>('/api/config/public', {

```typescript
import { useEffect, useMemo, useState } from 'react';
import {
  getPublicConfig,
  getSecretsStatus,
  getStatus,
  savePublicConfig,
  saveSecrets,
  startMonitor,
  stopMonitor,
} from '../api';
import {
  emptySecretsInput,
  emptySecretsStatus,
  emptyStatus,
  fallbackConfig,
  missingAccountFields,
  type SeatGroupMode,
} from '../configDefaults';
import type { EventItem, MonitorStatus, PublicConfig, SecretsPayload, SecretsStatus } from '../types';

export type BusyState = 'load' | 'save' | 'start' | 'stop' | null;

function isImportantEvent(item: EventItem) {
  return (
    item.message.includes('예약 완료') ||
    item.message.includes('취소표 발견') ||
    item.message.includes('자동 취소 실패') ||
    item.message.includes('예약 실패')
  );
}

function notifyBrowser(item: EventItem) {
  if (!('Notification' in window) || Notification.permission !== 'granted') return;
  new Notification('SRT Monitor', {
    body: item.message,
    tag: item.id,
  });
}

export function useMonitorController() {
  const [config, setConfig] = useState<PublicConfig>(fallbackConfig);
  const [secretsStatus, setSecretsStatus] = useState<SecretsStatus>(emptySecretsStatus);
  const [secretsInput, setSecretsInput] = useState<SecretsPayload>(emptySecretsInput);
  const [status, setStatus] = useState<MonitorStatus>(emptyStatus);
  const [events, setEvents] = useState<EventItem[]>([]);
  const [busy, setBusy] = useState<BusyState>('load');
  const [error, setError] = useState('');
  const [showAccountWarning, setShowAccountWarning] = useState(false);
  const [importantEvent, setImportantEvent] = useState<EventItem | null>(null);

  const passengerCount = useMemo(
    () =>
      Math.max(
        1,
        config.adult_count +
          config.child_count +
          config.senior_count +
          config.disability_1_to_3_count +
          config.disability_4_to_6_count,
      ),
    [config],
  );

  const seatGroupMode: SeatGroupMode = config.require_adjacent_seats
    ? 'adjacent'
    : config.require_same_car
      ? 'same-car'
      : 'any';

  const missingFields = missingAccountFields(secretsStatus, secretsInput);
  const accountMissingCount = missingFields.length;
  const accountReady = accountMissingCount === 0;

  // 초기 로드
  useEffect(() => {
    let ignore = false;

    async function load() {
      try {
        const [loadedConfig, loadedSecretsStatus, loadedStatus] = await Promise.all([
          getPublicConfig(),
          getSecretsStatus(),
          getStatus(),
        ]);
        if (!ignore) {
          const normalizedConfig = { ...fallbackConfig, ...loadedConfig };
          if (normalizedConfig.require_adjacent_seats) {
            normalizedConfig.require_same_row = false;
          }
          setConfig(normalizedConfig);
          setSecretsStatus(loadedSecretsStatus);
          setStatus(loadedStatus);
        }
      } catch (err) {
        if (!ignore) {
          setError(err instanceof Error ? err.message : '초기 데이터를 불러오지 못했습니다.');
        }
      } finally {
        if (!ignore) setBusy(null);
      }
    }

    load();
    return () => {
      ignore = true;
    };
  }, []);

  // SSE 이벤트 스트림
  useEffect(() => {
    const es = new EventSource('/api/events/stream');
    es.onmessage = (e) => {
      try {
        const item = JSON.parse(e.data) as EventItem;
        setEvents((prev) => {
          const next = [...prev, item];
          return next.length > 300 ? next.slice(-300) : next;
        });
        setStatus((current) => ({ ...current, last_message: item.message }));
        if (isImportantEvent(item)) {
          setImportantEvent(item);
          notifyBrowser(item);
        }
      } catch {
        // 잘못된 SSE 프레임 무시
      }
    };
    es.onerror = () => {
      setStatus((current) => ({ ...current, last_message: '실시간 이벤트 연결 확인 중' }));
    };
    return () => es.close();
  }, []);

  // 3초마다 상태 폴링
  useEffect(() => {
    const id = window.setInterval(async () => {
      try {
        setStatus(await getStatus());
      } catch {
        setStatus((current) => ({ ...current, last_message: '백엔드 연결 확인 중' }));
      }
    }, 3000);
    return () => window.clearInterval(id);
  }, []);

  function patchConfig(patch: Partial<PublicConfig>) {
    setConfig((current) => ({ ...current, ...patch }));
  }

  function patchSecretsInput(patch: Partial<SecretsPayload>) {
    setSecretsInput((prev) => {
      const next = { ...prev, ...patch };
      if (missingAccountFields(secretsStatus, next).length === 0) {
        setShowAccountWarning(false);
      }
      return next;
    });
  }

  function setPassengerCount(count: number) {
    patchConfig({
      adult_count: count,
      child_count: 0,
      senior_count: 0,
      disability_1_to_3_count: 0,
      disability_4_to_6_count: 0,
    });
  }

  function setSeatGroupMode(mode: SeatGroupMode) {
    patchConfig({
      require_same_row: false,
      require_same_car: mode === 'same-car',
      require_adjacent_seats: mode === 'adjacent',
    });
  }

  async function _saveSecretsIfNeeded(): Promise<void> {
    const hasNew = Object.values(secretsInput).some((v) => v.trim() !== '');
    if (!hasNew) return;
    const newStatus = await saveSecrets(secretsInput);
    setSecretsStatus(newStatus);
    setSecretsInput(emptySecretsInput);
  }

  async function handleSave() {
    setBusy('save');
    setError('');
    try {
      setConfig(await savePublicConfig(config));
      await _saveSecretsIfNeeded();
    } catch (err) {
      setError(err instanceof Error ? err.message : '설정 저장에 실패했습니다.');
    } finally {
      setBusy(null);
    }
  }

  async function handleStart() {
    setError('');
    if (missingFields.length > 0) {
      setShowAccountWarning(true);
      setError(`계정 정보가 전부 작성되지 않았습니다. 빠진 항목: ${missingFields.join(', ')}`);
      return;
    }
    setShowAccountWarning(false);
    if ('Notification' in window && Notification.permission === 'default') {
      await Notification.requestPermission();
    }
    setBusy('start');
    try {
      await _saveSecretsIfNeeded();
      setStatus(await startMonitor(config));
    } catch (err) {
      setError(err instanceof Error ? err.message : '모니터링 시작에 실패했습니다.');
    } finally {
      setBusy(null);
    }
  }

  async function handleStop() {
    setBusy('stop');
    setError('');
    try {
      setStatus(await stopMonitor());
    } catch (err) {
      setError(err instanceof Error ? err.message : '모니터링 중지에 실패했습니다.');
    } finally {
      setBusy(null);
    }
  }

  return {
    accountMissingCount,
    accountReady,
    busy,
    config,
    error,
    events,
    handleSave,
    handleStart,
    handleStop,
    importantEvent,
    passengerCount,
    patchConfig,
    patchSecretsInput,
    secretsInput,
    secretsStatus,
    setImportantEvent,
    seatGroupMode,
    setPassengerCount,
    setSeatGroupMode,
    setShowAccountWarning,
    showAccountWarning,
    status,
  };
}
```
